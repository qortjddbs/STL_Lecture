#include <random>
#include <print>		// C++ 23
#include <algorithm>
#include <array>		// C++ 11
#include "save.h"

std::default_random_engine dre;
std::uniform_int_distribution uid{ 0, 9999 };

// ----------
int main()
// ----------
{
	// [문제] 랜덤값을 갖는 int 1000개를 만들어라.
	// int[1000] 이런 자료구조는 앞으론 쓰지 말 것		new int[1000] 이거 쓸 때 말고는 안쓸거임 (이건 기계어 수준이라 안쓸 수 없음)
	// 대체할 수 있는 더 좋은 것이 있기 때문 - 시간/공간의 zero-overhead
	// 컴퓨터 세상에서 더 좋은 것 - 메모리를 덜 잡아먹음. 시간이 덜 걸림.
	// array<T, N>

	// int a[1000];		// 자료형은 int[N] 인데, 이름위치가 비정상

	std::array<int, 1000> a;

	for (int& num : a) {
		num = uid(dre);
	}

	for (int num : a) {
		std::print("{:8}", num);
	}
	std::cout << std::endl;

	// [문제] 가장 작은 값과 큰 값을 찾아서 화면에 출력하라.
	// syntatic sugar 인 structured binding을 사용하면 pair를 이렇게 쓸 수 있다!
	auto[작은값, 큰값] = std::minmax_element(std::begin(a), std::end(a));
	std::cout << "가장 작은 값: " << *작은값 << ", 가장 큰 값: " << *큰값 << std::endl;

	save("메인.cpp");
}
