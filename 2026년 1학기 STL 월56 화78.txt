// 여기 설명부터 다시 합니다.
// fname부터 하자

void save(const std::string& fname)
{
	std::ifstream in{ fname };
	std::ofstream out{ "2026년 1학기 STL 월56 화78.txt", std::ios::app };

	std::copy(std::istreambuf_iterator<char>{in}, {},
		std::ostream_iterator<char>{out});
}
